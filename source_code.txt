#include <Arduino.h>
#include <Wire.h>
#include <TinyGPSPlus.h>
#include <math.h>

#define SOS_BUTTON_PIN 4

#define MPU_SDA 20
#define MPU_SCL 21
#define MPU_ADDR 0x68

#define GPS_RX_PIN 5
#define GPS_TX_PIN 6
#define GPS_BAUD 9600

#define IMPACT_THRESHOLD 28000L
#define SOS_COOLDOWN_MS 5000UL
#define DEBOUNCE_DELAY_MS 50UL
#define TELEMETRY_INTERVAL_MS 2000UL

HardwareSerial GPSSerial(1);
TinyGPSPlus gps;

bool mpuAvailable = false;

unsigned long lastAlertTime = 0;
unsigned long lastTelemetryTime = 0;
unsigned long lastDebounceTime = 0;

int lastButtonReading = HIGH;
int stableButtonState = HIGH;

// Initialize MPU6050
bool initMPU() {
  Wire.begin(MPU_SDA, MPU_SCL);
  Wire.setClock(100000);

  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B);
  Wire.write(0x00);

  byte error = Wire.endTransmission(true);

  if (error == 0) {
    Serial.println("MPU6050_READY");
    return true;
  }

  Serial.println("MPU_ERROR");
  return false;
}

// Read accelerometer values
bool readMPU(int16_t &ax, int16_t &ay, int16_t &az) {
  if (!mpuAvailable) {
    return false;
  }

  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);

  if (Wire.endTransmission(false) != 0) {
    return false;
  }

  Wire.requestFrom((uint8_t)MPU_ADDR, (size_t)6, true);

  if (Wire.available() < 6) {
    return false;
  }

  ax = (int16_t)((Wire.read() << 8) | Wire.read());
  ay = (int16_t)((Wire.read() << 8) | Wire.read());
  az = (int16_t)((Wire.read() << 8) | Wire.read());

  return true;
}

// Send emergency event
void triggerAlert(const char *eventType) {
  unsigned long now = millis();

  if (lastAlertTime != 0 &&
      now - lastAlertTime < SOS_COOLDOWN_MS) {
    return;
  }

  lastAlertTime = now;
  Serial.println(eventType);
}

// Read GPS serial data
void readGPS() {
  while (GPSSerial.available()) {
    gps.encode(GPSSerial.read());
  }
}

// Send device telemetry as JSON
void sendTelemetry(int16_t ax, int16_t ay, int16_t az,
                   bool motionReadOK) {
  bool locationValid =
      gps.location.isValid() && gps.location.age() < 5000;

  Serial.print("{\"type\":\"telemetry\",\"battery\":100,\"gsm\":0");

  Serial.print(",\"ax\":");
  Serial.print(motionReadOK ? ax : 0);

  Serial.print(",\"ay\":");
  Serial.print(motionReadOK ? ay : 0);

  Serial.print(",\"az\":");
  Serial.print(motionReadOK ? az : 0);

  Serial.print(",\"gps_valid\":");
  Serial.print(locationValid ? "true" : "false");

  if (locationValid) {
    Serial.print(",\"latitude\":");
    Serial.print(gps.location.lat(), 6);

    Serial.print(",\"longitude\":");
    Serial.print(gps.location.lng(), 6);
  } else {
    Serial.print(",\"latitude\":null,\"longitude\":null");
  }

  Serial.println("}");
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(SOS_BUTTON_PIN, INPUT_PULLUP);

  mpuAvailable = initMPU();

  GPSSerial.begin(
      GPS_BAUD,
      SERIAL_8N1,
      GPS_RX_PIN,
      GPS_TX_PIN
  );

  Serial.println("SAFEBELT_USB_READY");
  Serial.println("GPS_UART_READY");
}

void loop() {
  unsigned long now = millis();

  // Read GPS data
  readGPS();

  // SOS button with debounce
  int reading = digitalRead(SOS_BUTTON_PIN);

  if (reading != lastButtonReading) {
    lastDebounceTime = now;
  }

  if (now - lastDebounceTime > DEBOUNCE_DELAY_MS) {
    if (reading != stableButtonState) {
      stableButtonState = reading;

      if (stableButtonState == LOW) {
        triggerAlert("SOS_BUTTON");
      }
    }
  }

  lastButtonReading = reading;

  // Read motion sensor
  int16_t ax = 0;
  int16_t ay = 0;
  int16_t az = 0;

  bool motionReadOK = readMPU(ax, ay, az);

  // Basic impact threshold detection
  if (motionReadOK) {
    long magnitude = (long)sqrt(
        (double)ax * ax +
        (double)ay * ay +
        (double)az * az
    );

    if (magnitude > IMPACT_THRESHOLD) {
      triggerAlert("FALL_DETECTED");
    }
  }

  // Send telemetry every 2 seconds
  if (now - lastTelemetryTime >= TELEMETRY_INTERVAL_MS) {
    lastTelemetryTime = now;

    sendTelemetry(ax, ay, az, motionReadOK);
  }

  delay(10);
}
